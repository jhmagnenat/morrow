export function generateLearningSection(documentId:string,options:GenerationOptions,splitBudget=1){
  const usage:LlmTelemetry={requests:0,inputTokens:0,outputTokens:0,elapsedMs:0,failures:0};
  return withLlmTelemetry(usage,()=>generate(documentId,options,usage,splitBudget));
}

async function generate(documentId:string,options:GenerationOptions,usage:LlmTelemetry,splitBudget:number){

  if(!document)throw Error('Document not found in this course');

  const textFor=(page:number)=>pageTexts.get(page)??'';
  const sourceFingerprint=digest(passages);
  const checkCurrent=()=>{
    if(options.claimId&&!ownsClaim(documentId,options.claimId))throw Error('Generation canceled');
    if(!getGraph().documents.some(d=>d.id===documentId)||digest(getPassagesForDocument(documentId))!==sourceFingerprint)throw Error('Generation canceled: source document changed');
  };
  const notify=(stage:string)=>{for(const page of pages)options.onProgress?.(page,stage);};

  if(imagePages.length){
    const {generateSlides}=await import('@/lib/generateSlides');

      result.notions.push(...fallback.notions);result.links.push(...(fallback.links??[]));result.cardsCreated+=fallback.cardsCreated;result.warnings.push(...fallback.warnings);

      result.processed+=fallback.passagesProcessed;
      if(fallback.warnings.some(w=>/quota|billing|invalid.*key|unauthoriz|request failed \((?:401|403)\)/i.test(w)))break;

    if(textPages.length&&!result.warnings.some(w=>/quota|billing|invalid.*key|unauthoriz|request failed \((?:401|403)\)/i.test(w))){

  const excluded=pages.filter(p=>nonTeachingSlideReason(textFor(p),insurance));
  checkCurrent();
  const db=openDb();
  for(const page of excluded){
    db.exec('BEGIN IMMEDIATE');
    try{

      // Old automatic logistics remain in the backup, outside learning queues.

    }).map(c=>c.key));

      const c=draft.concepts.find(c=>c.key===a.concept)!;

    });
    const reviewSources=snippets.map(s=>({...s,documentId}));
    const accuracy=await auditAccuracy(accuracyFields,reviewSources,{subject:draft.title,rule:'Hypothetical inputs belong only to explicitly hypothetical exercises. Check cross-concept contradictions.'});
    for(const [key,problems] of accuracy){const concept=assertions.find(a=>a.key===key)!.concept;issues.set(concept,[...(issues.get(concept)??[]),...problems]);}
    const verifiedIdentities=new Set([...identityCandidates].filter(key=>accuracyFields.some(f=>f.key===`${key}:definition`)&&!accuracy.has(`${key}:definition`)));
    const cardEligible=(concept:string)=>verifiedIdentities.has(concept)||!issues.get(concept)?.length&&audit.concepts.filter(d=>d.key===concept&&d.accepted).length===1;
    // A positive primary audit cannot override a rejected factual claim.
    // Review card questions AND answers, and every relationship independently.
    const recallFields:AccuracyField[]=[
      ...cards.filter(c=>cardEligible(c.concept)&&audit.cards.filter(d=>d.key===c.key&&d.accepted).length===1).map(c=>({key:`card:${c.key}`,text:`${c.question}\n${c.answer}`,kind:c.kind==='application'?'exercise' as const:'deduction' as const,sourceIds:cardSourceIds(c)})),
      ...draft.relations.filter(r=>audit.relations.filter(d=>d.key===r.key&&d.accepted).length===1).map(r=>({key:`relation:${r.key}`,text:r.explanation,kind:'deduction' as const,sourceIds:snippets.filter(s=>r.sources.some(e=>e.page===s.page&&e.quote===s.text)).map(s=>s.id)})),
    ];
    const recallAccuracy=await auditAccuracy(recallFields,reviewSources,{subject:draft.title});
    for(const [key,problems] of recallAccuracy){
      const list=key.startsWith('card:')?audit.cards:audit.relations;
      for(const decision of list.filter(d=>d.key===key.slice(key.indexOf(':')+1))){decision.accepted=false;decision.reason=problems.join('; ');}
    }
    checkCurrent();
    const repairRequests:CardRepairRequest[]=cards.flatMap(c=>{
      if(!cardEligible(c.concept))return [];
      const decisions=audit.cards.filter(d=>d.key===c.key),decision=decisions.length===1?decisions[0]:undefined;
      // Missing audit data is not actionable pedagogical feedback.
      if(!decision)return [];
      const problems=[...cardQuality(c.question,c.answer,c.kind),...(recallAccuracy.get(`card:${c.key}`)??[]),...(!decision.accepted&&decision.reason?[decision.reason]:[])];
      if(!problems.length)return [];
      return [{card:{...c,sourceIds:cardSourceIds(c)},issues:problems}];
    });
    if(repairRequests.length){
      notify('Improving recall cards from reviewer feedback');
      try{
        const repair=await repairCourseCards(repairRequests,reviewSources,{subject:draft.title,concepts:draft.concepts.map(c=>({key:c.key,label:c.label}))});
        checkCurrent();
        for(const revised of repair.cards){
          const index=cards.findIndex(c=>c.key===revised.key),decision=audit.cards.find(d=>d.key===revised.key)!;
          cards[index]={...cards[index],question:revised.question,answer:revised.answer};
          decision.accepted=true;decision.reason='';decision.duplicateOf='';
        }
        result.warnings.push(...repair.warnings);
      }catch(error){checkCurrent();result.warnings.push(`Card correction could not be verified: ${error instanceof Error?error.message:String(error)}`);}
    }
    const exactDecisions=(expected:string[],actual:Array<{key:string;accepted:boolean}>)=>actual.length===expected.length&&expected.every(key=>actual.filter(d=>d.key===key&&d.accepted).length===1);
    const reusableAudit=primary.every(page=>audit.pages.filter(p=>p.page===page&&p.complete&&!p.excluded&&!p.missing.length).length===1)&&audit.pages.length===primary.length&&[...issues.values()].every(i=>!i.length)&&exactDecisions(keys,audit.concepts)&&exactDecisions(cards.map(c=>c.key),audit.cards)&&exactDecisions(relationKeys,audit.relations)&&cards.every(c=>!cardQuality(c.question,c.answer,c.kind).length);
    const completeDecisions=(expected:string[],actual:Array<{key:string}>)=>actual.length===expected.length&&expected.every(key=>actual.filter(d=>d.key===key).length===1);
    const completeRejection=primary.every(page=>audit.pages.filter(p=>p.page===page).length===1)&&audit.pages.length===primary.length&&completeDecisions(keys,audit.concepts)&&completeDecisions(cards.map(c=>c.key),audit.cards)&&completeDecisions(relationKeys,audit.relations)&&completeDecisions(assertions.map(a=>a.key),audit.assertions??[])&&(audit.assertions??[]).every(d=>!d.supported||d.pages.length>0&&d.pages.every(page=>assertions.find(a=>a.key===d.key)?.pages.includes(page)));
    // Reuse a complete rejection too: identical material must not repeatedly
    // incur paid approvals or get a lucky pass from stochastic re-auditing.
    if(!savedAudit&&(reusableAudit||completeRejection))db.prepare('INSERT INTO quality_settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value').run(auditKey,JSON.stringify(audit));
    notify('Saving checked learning section');
    const one=<T extends {key:string}>(list:T[],key:string)=>{const found=list.filter(d=>d.key===key);return found.length===1?found[0]:undefined;};
    const incompleteLessons=new Set<string>(),resolved=new Map<string,string>(),savedCards=new Map<string,StudyCard[]>(),rejectedCards=new Set<string>(),rejectedRelations:CoreDraft['relations']=[];
    db.exec('BEGIN IMMEDIATE');
    try{
      const graph=getGraph();
      for(const previous of relatedNotes){
        const current=graph.notions.find(n=>n.id===previous.id);
        if(!current||current.definition!==previous.definition||(current.explanation??'').slice(0,1200)!==previous.explanation)throw Error('Related notes changed during audit; retry this section against the current course.');
      }
      for(const candidate of draft.concepts){
        const concurrent=graph.notions.find(n=>canonicalLabel(n.label)===canonicalLabel(candidate.label)&&n.sources.some(s=>s.documentId===documentId)&&!relatedNotes.some(previous=>previous.id===n.id));
        if(concurrent&&concurrent.definition!==candidate.definition)throw Error('Related notes changed during audit; a concurrently created definition needs a consistency check.');
      }
      for(const c of draft.concepts){
        const decision=one(audit.concepts,c.key),localIssues=issues.get(c.key)!;
        if(!c.lesson||!decision?.accepted||localIssues.length){
          if(!decision?.accepted||localIssues.length)result.warnings.push(`${c.label}: ${localIssues.join('; ')||decision?.reason||'Missing independent audit'}`);
          if(!verifiedIdentities.has(c.key))continue;
          // Retain the independently reviewed identity without publishing any
          // rejected explanatory prose or creating invented lesson activities.
          const e=proof.get(c.key)!;
          let identity=graph.notions.find(n=>canonicalLabel(n.label)===canonicalLabel(c.label)&&canonicalLabel(n.definition)===canonicalLabel(c.definition)&&n.sources.some(s=>s.documentId===documentId));
          if(!identity){

          }

      }
      db.exec('COMMIT');
    }catch(error){db.exec('ROLLBACK');throw error;}
    result.warnings=[...new Set(result.warnings)];usage.requests=Math.max(requests,usage.requests);finishRun(result.warnings.length?'needs_review':'complete');return result;
  }catch(error){
    const reason=error instanceof Error?error.message:'Section generation failed';result.warnings.push(reason);usage.requests=Math.max(requests,usage.requests);
    if(/canceled|source document changed/i.test(reason))throw error;

    if(!/Related notes changed during audit/i.test(reason))db.prepare('DELETE FROM quality_settings WHERE key=?').run(cacheKey);
    for(const page of primary)savePageAnalysis(documentId,{page,kind:'content',status:'retry',reason,warnings:[reason],concepts:0,cards:0,missingObjectives:[reason],updatedAt:new Date().toISOString(),version:CORE_PIPELINE_VERSION});
    finishRun('needs_review');
    // A bounded output failure can be repaired by two smaller sections once.
    // Source/semantic rejection never starts an automatic paid repair loop.
    if(splitBudget>0&&primary.length>1&&/truncated|smaller page batch|exceed.*twelve/i.test(reason)){
      result.warnings=[];
      const middle=Math.ceil(primary.length/2);
      for(const smaller of [primary.slice(0,middle),primary.slice(middle)]){
        checkCurrent();
        const repaired=await generateLearningSection(documentId,{...options,pages:smaller},splitBudget-1);
        result.notions.push(...repaired.notions);result.links.push(...repaired.links);result.cardsCreated+=repaired.cardsCreated;result.readyCards+=repaired.readyCards;result.processed+=repaired.processed;result.warnings.push(...repaired.warnings);
        usage.requests+=repaired.usage.requests;usage.inputTokens+=repaired.usage.inputTokens;usage.outputTokens+=repaired.usage.outputTokens;usage.elapsedMs+=repaired.usage.elapsedMs;usage.failures+=repaired.usage.failures;
        if(repaired.warnings.some(w=>/quota|billing|invalid.*key|unauthoriz/i.test(w)))break;
      }
    }
    return result;
  }
