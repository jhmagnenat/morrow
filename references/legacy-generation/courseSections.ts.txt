// The model writes content and citations. Internal keys and duplicated note
// identities are assigned here, so a typo cannot invalidate a whole lesson.
export function normalizeSectionDraft(draft:SectionDraft,sources:ReturnType<typeof sourceSnippets>,primaryPages?:number[]) {
  if(primaryPages)draft.pages=primaryPages.map(page=>draft.pages.find(p=>p.page===page)??{page,kind:'content',reason:'Primary source page; subject coverage requires independent audit.',teachingPoints:[],conceptKeys:[]});
  const aliases=new Map<string,string[]>();
  const addAlias=(alias:string,key:string)=>{if(!alias)return;const name=canonicalLabel(alias),ids=aliases.get(name)??[];if(!ids.includes(key))ids.push(key);aliases.set(name,ids);};
  draft.concepts.forEach((c,index)=>{const old=c.key;c.key=`C${index+1}`;addAlias(old,c.key);addAlias(c.label,c.key);c.label=c.label.replace(/([a-z])([A-Z])/g,'$1 $2').replace(/([A-Z]+)([A-Z][a-z])/g,'$1 $2');addAlias(c.label,c.key);c.note.key=c.key;c.note.branchKey=c.branchKey!;c.evidence=[...c.evidence??[],...c.note.evidence].filter((e,i,a)=>a.findIndex(v=>v.sourceId===e.sourceId)===i);});
  // Facts live inside their note in the model response. The server derives
  // page inventories and identities, with no cross-note label references.
  for(const page of draft.pages)page.teachingPoints=[];
  for(const c of draft.concepts){
    if(c.note.learning){
      try{
        c.note.explanation=composeIntuitiveLesson(c.note.learning,c.note.sketch.relations);
        c.note.sketch={relations:[],example:''};
      }catch{c.note.explanation='';} // Named validation below gets one bounded repair.
    }
    c.note.teachingPoints=[];
    for(const [index,detail] of (c.note.sourceDetails??[]).entries()){
      const source=sources.find(s=>s.id===detail.sourceId);
      if(!source)throw new Error('Section detail cites an unknown source');
      const page=draft.pages.find(p=>p.page===source.page);
      if(!page)continue; // Reference facts clarify notes, not primary coverage.
      const key=`${c.key}P${page.page}T${index+1}`;
      page.teachingPoints.push({...detail,key,quote:source.quote});c.note.teachingPoints.push(key);
      if(!c.evidence?.some(e=>e.sourceId===detail.sourceId))c.evidence?.push({sourceId:detail.sourceId});
      if(!c.note.evidence.some(e=>e.sourceId===detail.sourceId))c.note.evidence.push({sourceId:detail.sourceId});
    }
    c.teachingPoints=c.note.teachingPoints;
    const retained=(c.note.sourceDetails??[]).map(point=>point.detail.trim()).filter(detail=>detail && !c.note.explanation.includes(detail));
    if(retained.length)c.note.explanation+='\n\nDetails retained from the slide:\n\n'+[...new Set(retained)].map(detail=>`\u2022 ${detail}`).join('\n\n');
  }
  for(const page of draft.pages)page.conceptKeys=draft.concepts.filter(c=>c.evidence?.some(e=>sources.find(s=>s.id===e.sourceId)?.page===page.page)).map(c=>c.key);
  const resolve=(value:string)=>{const matches=aliases.get(canonicalLabel(value));return matches?.length===1?matches[0]:undefined;};
  // A malformed optional edge is rejected individually. It cannot erase valid
  // source notes/cards. Required source relationships remain in the inventory
  // and their coverage is still independently checked against the explanation.
  draft.links=draft.links.flatMap((link,index)=>{const source=resolve(link.source),target=resolve(link.target);return source&&target&&source!==target?[{...link,key:`L${index+1}`,source,target}]:[];});
  return draft;
}

// Stable groups follow the subject plan, including non-adjacent slides. A
// bounded group keeps vision/output manageable; the rest of its theme remains
// reference context, rather than being mistaken for another lesson.
export function sectionPages(page:number,plan:CourseLearningPlan,passages:SectionInput['passages'],insurance:boolean) {
  const available=[...new Set(passages.map(p=>p.page))].sort((a,b)=>a-b).filter(p=>!nonTeachingSlideReason(passages.filter(s=>s.page===p).map(s=>s.text).join('\n'),insurance));
  const branch=plan.branches.find(b=>b.pages.includes(page));
  // An unassigned page is examined on its own. Combining unrelated leftovers
  // would recreate the very glossary fragmentation this module prevents.
  const assigned=branch ? available.filter(p=>branch.pages.includes(p)) : [page];
  const index=assigned.indexOf(page),start=Math.floor(Math.max(0,index)/3)*3;
  const group=assigned.slice(start,start+3);
  return {pages:group.includes(page)?group:[page],referencePages:branch?.pages??[page],branch};
}

}

export function validateSection(draft:SectionDraft,pages:number[],sources:ReturnType<typeof sourceSnippets>,plan:CourseLearningPlan,imagePages:number[]) {
  if(draft.pages.length!==pages.length || pages.some(page=>draft.pages.filter(p=>p.page===page).length!==1))throw new Error('Section must inventory every primary page exactly once');
  const keys=new Set(draft.concepts.map(c=>c.key)),sourceIds=new Set(sources.map(s=>s.id));
  if(keys.size!==draft.concepts.length || draft.concepts.some(c=>!c.key.trim()))throw new Error('Section concept keys must be unique');
  const points=draft.pages.flatMap(p=>p.teachingPoints);
  if(new Set(points.map(p=>p.key)).size!==points.length)throw new Error('Section teaching-point keys must be unique');
  for(const page of draft.pages){
    if(['content','exercise'].includes(page.kind) && !page.teachingPoints.length)throw new Error(`Missing source-detail note for page ${page.page}`);
    if(page.conceptKeys.some(key=>!keys.has(key)))throw new Error('Section page references an unknown concept');
    for(const point of page.teachingPoints){
      const source=sources.find(s=>s.id===point.sourceId);
      if(!(source?.page===page.page && source.quote) && !(imagePages.includes(page.page) && point.visualDescription.trim().length>20))throw new Error('Section point lacks primary-page evidence');
    }
  }
  for(const c of draft.concepts){
    if(!c.note.learning)throw new Error('Missing intuitive learning model');
    composeIntuitiveLesson(c.note.learning);
    if(c.note.key!==c.key || c.branchKey!==c.note.branchKey || !plan.branches.some(b=>b.key===c.branchKey))throw new Error('Section note has inconsistent identity or branch');
    if(!c.label.trim() || !c.definition.trim())throw new Error('Section concept lacks a definition');
    if([...c.evidence??[],...c.note.evidence].some(e=>!sourceIds.has(e.sourceId??'')))throw new Error('Section cites an unknown source');
    if(c.note.teachingPoints.some(key=>!points.some(p=>p.key===key)))throw new Error('Section note references an unknown teaching point');
    prepareTeachingBridges(c.key,c.note.bridges,sourceIds);
    composeLearningSketch(c.note.sketch,true);composeLearningExplanation(c.note.explanation,c.note.workflow,c.note.confusions,true);
  }
  if(new Set(draft.links.map(l=>l.key)).size!==draft.links.length || draft.links.some(l=>!keys.has(l.source)||!keys.has(l.target)||l.evidence?.some(e=>!sourceIds.has(e.sourceId??''))))throw new Error('Section relationship has invalid endpoints or evidence');
}
