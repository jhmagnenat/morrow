// Pure educational engine: source inventory, fixed-target writer and independent review; one bounded repair round.
// It cannot read SQLite, inspect the graph, write cards or alter scheduling.
export async function runGenerationEngine(source:GenerationSource,existing:StudyCard[],model:GenerationModel,detail:'key-ideas'|'comprehensive'='key-ideas',repair=false,requested?:DirectDraft['knowledge']):Promise<GenerationOutput>{
  const evidence={document:source.title,primaryPages:source.primaryPages,courseOutline:source.courseOutline??'',layouts:source.layouts??[],
    task:'Teach the substantive knowledge in primary sources. Context sources clarify primary material; they are not additional targets for card generation.',
    sources:source.snippets.map(snippet=>({...snippet,role:source.primaryPages.includes(snippet.page)?'primary':'context'})),
    imagePages:source.imagePages.map((page,index)=>({page,role:source.primaryPages.includes(page)?'primary':'context',sourceId:`p${page}image`,imageIndex:index}))};
  // Existing generated answers never enter the writer's source context, and
  // saving cards cannot invalidate the writer's immutable-source cache.
  const diagnostics:string[]=[],knowledge:DirectDraft['knowledge']=[],candidates:DirectCard[]=[],seenQuestions=new Set<string>();
  if(requested)knowledge.push(...requested.map(k=>({...k,recall:undefined,selected:true,cardKeys:[],repairOf:k.key})));
  else{
    const extracted=source.snippets.filter(s=>source.primaryPages.includes(s.page)).flatMap(s=>productFamilyTableTargets(s,source.layouts?.find(l=>l.page===s.page))??balanceSheetFunctionTargets(s)??positionedRoleTargets(s,source.layouts?.find(l=>l.page===s.page))??actorContributionTargets(s)??insuranceEconomyTargets(s)??insuranceChallengeTargets(s)??insuranceCompanyTargets(s)??insuranceCompanyCategories(s)??[]);
    const extractedPages=new Set(extracted.map(t=>t.page));
    const modelPages=source.primaryPages.filter(p=>!extractedPages.has(p));
    const sparseImages=source.imagePages.filter(page=>modelPages.includes(page)||!source.primaryPages.includes(page));
    const planIds=[...source.snippets.map(s=>s.id),...sparseImages.map(p=>`p${p}image`)];
    const requiredQuestions=modelPages.flatMap(page=>knowledgeCheckQuestions({page,text:source.snippets.filter(s=>s.page===page).map(s=>s.text).join('')}));
    const planSchema=directPlanSchema(planIds,modelPages.length?modelPages:source.primaryPages,requiredQuestions);
    const rawPlan=modelPages.length?await model.complete<DirectPlanSlots>(PLAN,{...evidence,imagePages:sparseImages.map((page,imageIndex)=>({page,role:source.primaryPages.includes(page)?'primary':'context',sourceId:`p${page}image`,imageIndex})),primaryPages:modelPages,requiredQuestions,sources:evidence.sources.filter(s=>s.role==='context'||modelPages.includes(s.page)),coverage:detail},'plan_direct_cards','Extract independently testable source knowledge',planSchema,sparseImages.map(p=>source.images[source.imagePages.indexOf(p)] )):{targets:[],questions:{}};
    const invalid=findSchemaViolations(rawPlan,planSchema);if(invalid.length)throw Error(`Invalid direct card plan: ${invalid.join('; ')}`);
    const planned={targets:[...rawPlan.targets,...Object.entries(rawPlan.questions).map(([key,value])=>({...value,importance:'core' as const,page:requiredQuestions.find(q=>q.key===key)!.page}))].filter((t,index,all)=>all.findIndex(other=>other.page===t.page&&other.claim===t.claim)===index)};
    for(const target of planned.targets)if(Object.values(rawPlan.questions).some(q=>q.claim&&q.claim===target.claim))target.importance='core';
    // Inventory and omission detection must work identically after a reset.
    // The independent source audit, never an old deck, supplies missing goals.
    const plan={targets:[...extracted,...planned.targets]};
    plan.targets.forEach((target,index)=>{
      const issues:string[]=[];
      if(!target.claim.trim())issues.push('Target has no learning claim');
      if(sourceQuestionClaim(target.claim))issues.push('An unanswered source question is not an established answer');
      if(/^(?:understand|identify|recognize|learn|determine|know|explain|comprendre|identifier|reconnaitre|apprendre)\b/i.test(target.claim.trim()))issues.push('A learning instruction is not a source-established factual claim');
      if(!target.sourceIds.some(id=>source.snippets.some(s=>s.id===id&&s.page===target.page)||id===`p${target.page}image`))issues.push('Target has no primary source');
      diagnostics.push(...issues.map(i=>`t${index}: ${i}`));
      const selected=detail==='comprehensive'?target.importance!=='incidental':!target.importance||target.importance==='core';
      knowledge.push({...target,selected,key:`t${index}`,cardKeys:[],issues});
    });
  }
  for(const target of knowledge)if(/^(?:understand|identify|recognize|learn|determine|know|explain|comprendre|identifier|reconnaitre|apprendre)\b/i.test(target.claim.trim())&&!target.issues?.length)target.issues=['A learning instruction is not a source-established factual claim'];
  for(const target of knowledge)if(sourceQuestionClaim(target.claim))target.issues=[...new Set([...(target.issues??[]),'An unanswered source question is not an established answer'])];
  const eligible=knowledge.filter(k=>k.selected!==false&&!k.issues?.length&&!k.exclusionReason);
  const generative=eligible.filter(k=>!k.recall);
  const usedImages=source.imagePages.filter(page=>knowledge.some(k=>k.sourceIds.includes(`p${page}image`))||!hasSourceTeachingText(source.snippets.filter(s=>s.page===page).map(s=>s.text).join('\n')));
  const scopedIds=[...source.snippets.map(s=>s.id),...usedImages.map(p=>`p${p}image`)];
  const scopedImages=usedImages.map(p=>source.images[source.imagePages.indexOf(p)]);
  const scopedEvidence={...evidence,imagePages:usedImages.map((page,index)=>({page,role:source.primaryPages.includes(page)?'primary':'context',sourceId:`p${page}image`,imageIndex:index}))};
  const draftSchema=directDraftSchema(scopedIds,generative.length,generative.map(k=>k.key));
  const rawDraft=generative.length?await model.complete<TargetCards>(DESIGN,{...scopedEvidence,coverage:detail,knowledge:generative,repairTargets:requested??[]},'direct_cards','Write recall cards for fixed source-grounded targets',draftSchema,scopedImages):{cards:{}};
  const malformed=findSchemaViolations(rawDraft,draftSchema);if(malformed.length)throw Error(`Invalid direct card output: ${malformed.join('; ')}`);
  for(const target of eligible.filter(k=>k.recall))rawDraft.cards[target.key]={...target.recall!,key:target.key,topicDefinition:target.claim,explanation:'',kind:target.recall?.kind??'mechanism',objective:target.claim,sources:target.sourceIds.map(sourceId=>({sourceId})),visualDescription:''};
  for(const [targetKey,card] of Object.entries(rawDraft.cards)){
    const target=knowledge.find(k=>k.key===targetKey);if(!target)continue;
    const relation=target.claim.match(/^(.+?) (?:is|are) (?:a function |functions? )?(?:associated with|related to) (.+?)(?: on the balance sheet)?[.]?$/i);
    const question=(target.recall?.question??(relation?`Which category is associated with ${relation[1]}${/on the balance sheet/i.test(target.claim)?' on the balance sheet':''}?`:card.question)).normalize('NFC').replace(/\s+/g,' ').trim();
    if(seenQuestions.has(question)){diagnostics.push(`Repeated question for ${target.key}; retained only its first occurrence`);continue;}
    seenQuestions.add(question);
    const identity=`${requested?'repair-':''}${target.key}c${target.cardKeys.length}`;
    // The extracted fact, not a generic writer label, identifies the objective.
    // Distinct functions related to the same category must not overwrite each other.
    // Bind explanatory content to the independently reviewed source claim.
    // Rewording a label into a textbook definition was introducing facts absent
    // from sparse slides despite the writer and reviewer instructions.
    const explanation=card.explanation?.trim()??'';
    const sourceExplanation=explanation&&source.snippets.some(s=>card.sources.some(e=>e.sourceId===s.id)&&verifiesQuote(explanation,s.text))?explanation:'';
    // Simple association facts have a direct recall direction. Asking for all
    // members but answering with one member created ambiguous, incomplete cards.
    const answer=target.recall?.answer??(relation?relation[2].replace(/[.]$/,'')+'.':card.answer);
    candidates.push({...card,question,answer,key:identity,objective:target.claim,topicDefinition:target.claim,explanation:sourceExplanation});target.cardKeys.push(identity);
  }
  const excludedCards=candidates.flatMap(c=>{const reason=administrativeContentReason(`${c.topic} ${c.question} ${c.objective}`);return reason?[{key:c.key,reason}]:[];});
  const draft:DirectDraft={knowledge,cards:candidates.filter(c=>!excludedCards.some(excluded=>excluded.key===c.key))};
  const contracts=new Map(draft.cards.map(c=>[c.key,directCardIssues(c,draft.cards,source.snippets,source.imagePages,source.primaryPages)]));
  const reviewable=draft.cards.filter(c=>!contracts.get(c.key)?.length),fields=reviewable.map(directField);
  // Source/design review is independent of mutable deck contents. Exact
  // objective reuse and protected-card checks happen in the transaction.
  // Adding a card must not invalidate an already reviewed immutable field.
  const requiredQuestions=source.primaryPages.flatMap(page=>knowledgeCheckQuestions({page,text:source.snippets.filter(s=>s.page===page).map(s=>s.text).join('')}));
  const reviewSchema=directReviewSchema(reviewable,source.primaryPages,scopedIds,[],knowledge,requiredQuestions);
  const rawResponse=await model.complete<DirectReviewSlots>(REVIEW,{...scopedEvidence,coverage:detail,reviewContract:'fixed-fields-v10-source-only-cache',requiredQuestions,repairTargets:requested??[],knowledge:draft.knowledge,targetCards:draft.knowledge.map(k=>({key:k.key,claim:k.claim,cards:draft.cards.filter(c=>k.cardKeys.includes(c.key)).map(c=>({key:c.key,question:c.question,answer:c.answer}))})),fields,cards:reviewable.map(c=>({key:c.key,objective:c.objective,kind:c.kind})),existingCards:[]},'audit_direct_cards','Check claims, recall quality and source coverage',reviewSchema,scopedImages);
  const invalid=findSchemaViolations(rawResponse,reviewSchema);if(invalid.length)throw Error(`Invalid direct card review: ${invalid.join('; ')}`);
  const response=unpackDirectReview(rawResponse,requiredQuestions);
  const review:DirectReview={targets:response.targets,decisions:response.decisions.map(({verdict,evidence,reasoning,...decision})=>({...decision,claims:[{text:fields.find(f=>f.key===decision.key)?.text??'',verdict,evidence:evidence.map(e=>({...e,quote:source.snippets.find(s=>s.id===e.sourceId)?.text??''})),reasoning}]})),pages:response.pages.map(p=>({...p,excluded:p.excluded&&!source.snippets.some(s=>s.page===p.page&&isKnowledgeCheck(s.text)),objectives:p.objectives.filter(o=>!administrativeContentReason(o.description))}))};
  // Text is checked against exact original passages. Image-only publication
  // additionally requires an independent read-back of the original image.
  const textualIds=new Set(source.snippets.map(s=>s.id));
  // Bind text-only verdicts to original source IDs. The model returns IDs,
  // never retyped source text: this avoids PDF line/page-number artifacts.
  for(const decision of review.decisions)if(!decision.requiresVisualEvidence)for(const claim of decision.claims){
    claim.evidence=claim.evidence.filter(e=>textualIds.has(e.sourceId));
  }
  const visualCandidates=reviewable.filter(c=>{
    const decision=review.decisions.find(d=>d.key===c.key),target=knowledge.find(k=>k.cardKeys.includes(c.key)),checked=review.targets.find(t=>t.key===target?.key);
    return decision?.requiresVisualEvidence&&decision.complete&&decision.scopeCorrect&&decision.atomic&&decision.unambiguous&&decision.relevant&&decision.claims.every(claim=>claim.verdict==='supported')&&checked?.verdict==='supported'&&checked.atomic&&checked.coverageComplete&&checked.selectionCorrect!==false&&c.sources.some(s=>usedImages.some(p=>s.sourceId===`p${p}image`));
  });
  let visualConfirmations:Awaited<ReturnType<typeof confirmVisualCandidates>>={};
  try{visualConfirmations=await confirmVisualCandidates(source,visualCandidates,model);}catch(error){diagnostics.push(`Independent image check failed; textual cards preserved: ${error instanceof Error?error.message:String(error)}`);}
  const factual=accuracyIssues(fields.filter(f=>!visualConfirmations[f.key]).map(f=>({...f,sourceIds:f.sourceIds.filter(id=>textualIds.has(id))})),source.snippets,review);
  const cards=draft.cards.map(candidate=>{
    const contract=contracts.get(candidate.key)!,issues=[...contract];
    const target=knowledge.find(k=>k.cardKeys.includes(candidate.key));
    const targetMatches=review.targets.filter(t=>t.key===target?.key),targetReview=targetMatches.length===1?targetMatches[0]:undefined;
    if(!targetReview||targetReview.verdict!=='supported'||!targetReview.atomic||targetReview.selectionCorrect===false)issues.push(targetReview?.reason||'Missing or rejected target review');
    const decisions=review.decisions.filter(d=>d.key===candidate.key),decision=decisions.length===1?decisions[0]:undefined;
    const visual=visualConfirmations[candidate.key];
    const checked=decision?.claims.flatMap(c=>c.evidence)??[];
    const textOnly=!decision?.requiresVisualEvidence&&checked.length>0&&checked.every(e=>textualIds.has(e.sourceId));
    const selected=visual?[...visual.sourceIds.map(sourceId=>({sourceId})),...candidate.sources.filter(s=>source.snippets.some(t=>t.id===s.sourceId&&source.primaryPages.includes(t.page)&&isKnowledgeCheck(t.text)))]:textOnly?checked:candidate.sources;
    const evidence:Evidence[]=selected.flatMap(({sourceId})=>{
      const snippet=source.snippets.find(s=>s.id===sourceId);
      if(snippet)return [{documentId:source.documentId,page:snippet.page,quote:snippet.text}];
      const page=source.imagePages.find(p=>sourceId===`p${p}image`);
      const description=visual?.observedEvidence??candidate.visualDescription;
      return page===undefined?[]:[{documentId:source.documentId,page,quote:'',visualDescription:description,...(visual?{visualReview:{reviewer:`Automatic independent image check (${source.reviewerLabel??'configured reviewer'})`,reviewedAt:new Date().toISOString(),pdfSha256:source.pdfSha256!,page,description,scope:'visible-facts-only' as const,recallHash:recallContentHash({...candidate,example:''})}}:{})}];
    });
    if(!contract.length)issues.push(...(factual.get(candidate.key)??[]));
    if(decision?.requiresVisualEvidence&&!visual)issues.push('Image-supported answer requires visual review against the original PDF');
    if(target&&!visual&&!hasSourceTeachingText(source.snippets.filter(s=>s.page===target.page).map(s=>s.text).join('\n')))issues.push('Primary page contains only a heading or reference; its factual assertions require original-image review');
    if(!decision||!decision.atomic||!decision.unambiguous||!decision.relevant)issues.push(decision?.reason||'Missing or rejected card design review');
    if(!evidence.some(e=>source.primaryPages.includes(e.page)))issues.push('No reviewed primary-page evidence');
    if(evidence.some(e=>!e.quote&&!e.visualReview))issues.push('Image-supported answer requires visual review against the original PDF');
    if(!visual&&evidence.some(e=>source.uncertainOcrPages.includes(e.page)))issues.push('Uncertain OCR evidence requires review');
    return {candidate,evidence,issues:[...new Set(issues)],contractValid:!contract.length&&targetReview?.verdict!=='non-teaching'};
  });
  const output:GenerationOutput={draft,review,cards,excludedCards,diagnostics};
  return repair?repairGeneration(source,existing,model,detail,output):output;
}

// One bounded repair round. Successful cards survive a failed repair call.
async function repairGeneration(source:GenerationSource,existing:StudyCard[],model:GenerationModel,detail:'key-ideas'|'comprehensive',output:GenerationOutput):Promise<GenerationOutput>{
  const unresolved=output.draft.knowledge.filter(k=>{
    const review=output.review.targets.find(t=>t.key===k.key);
    const visual=k.cardKeys.some(key=>output.review.decisions.some(d=>d.key===key&&d.requiresVisualEvidence));
    const text=k.sourceIds.some(id=>source.snippets.some(s=>s.id===id&&s.text.trim().length>=12&&!source.uncertainOcrPages.includes(s.page)));
    if(k.selected===false)return text&&!k.issues?.length&&review?.selectionCorrect===false&&review.verdict==='supported'&&review.atomic;
    return text&&!visual&&!k.issues?.length&&review?.verdict==='supported'&&review.atomic&&(!review.coverageComplete||!k.cardKeys.some(key=>output.cards.some(c=>c.candidate.key===key&&!c.issues.length)));
  });
  for(const page of output.review.pages)for(const objective of page.objectives)if(!objective.description.startsWith('Knowledge check:')&&!objective.cardKeys.length&&!unresolved.some(k=>k.page===page.page&&k.claim===objective.description)&&!output.draft.knowledge.some(k=>k.page===page.page&&k.claim===objective.description&&k.selected===false&&output.review.targets.some(t=>t.key===k.key&&t.selectionCorrect===true))){
    const target={key:`missing-${page.page}-${unresolved.length}`,page:page.page,claim:objective.description,sourceIds:source.snippets.filter(s=>s.page===page.page).map(s=>s.id),cardKeys:[],exclusionReason:''};
    if(target.sourceIds.length){unresolved.push(target);output.draft.knowledge.push(target);}
  }
  const selected=unresolved.slice(0,6);if(!selected.length)return output;
  try{
    const fixed=await runGenerationEngine(source,existing,model,detail,false,selected);
    output.repairs=[fixed];
    for(const target of selected){
      const unit=fixed.draft.knowledge.find(k=>k.repairOf===target.key&&!k.issues?.length);
      const checked=fixed.review.targets.find(t=>t.key===unit?.key);
      const accepted=fixed.cards.filter(c=>unit?.cardKeys.includes(c.candidate.key)&&!c.issues.length);
      if(!unit||checked?.verdict!=='supported'||!checked.atomic||!checked.coverageComplete||!accepted.length)continue;
      const oldKeys=new Set(target.cardKeys);
      output.cards=output.cards.filter(c=>!oldKeys.has(c.candidate.key));output.cards.push(...accepted);
      output.draft.cards=output.draft.cards.filter(c=>!oldKeys.has(c.key));output.draft.cards.push(...accepted.map(c=>c.candidate));
      target.cardKeys=accepted.map(c=>c.candidate.key);
      target.selected=true;
      output.review.decisions=output.review.decisions.filter(d=>!oldKeys.has(d.key));output.review.decisions.push(...fixed.review.decisions.filter(d=>target.cardKeys.includes(d.key)));
      output.review.targets=output.review.targets.filter(t=>t.key!==target.key);output.review.targets.push({...checked,key:target.key});
      for(const page of output.review.pages)if(page.page===target.page){
        for(const objective of page.objectives)if(objective.description===target.claim||objective.cardKeys.some(key=>oldKeys.has(key)))objective.cardKeys=[...new Set([...objective.cardKeys.filter(key=>!oldKeys.has(key)),...target.cardKeys])];
      }
    }
    output.diagnostics?.push(`Bounded repair reviewed ${selected.length} targets; no second repair round`);
  }catch(error){
    const reason=error instanceof Error?error.message:String(error);
    if(/canceled|source document changed/i.test(reason))throw error;
    output.diagnostics?.push(`Targeted repair failed; initial accepted cards retained: ${reason}`);
  }
  return output;
}
